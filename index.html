Institutional Pine Script v5 Strategies

//@version=5 strategy("AI Claude - VWAP Trend Pullback Strategy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=10) fast_ema = ta.ema(close, 9) slow_ema = ta.ema(close, 21) trend_ema = ta.ema(close, 200) vwap_val = ta.vwap(hlc3) atr_val = ta.atr(14) bullish_trend = (close > trend_ema) and (fast_ema > slow_ema) pullback_to_vwap = (low <= vwap_val) and (close >= vwap_val) long_entry = bullish_trend and pullback_to_vwap and (close > open) if (long_entry and strategy.position_size == 0) stop_level = close - (atr_val * 1.5) target_level = close + ((close - stop_level) * 2.0) strategy.entry("Long", strategy.long) strategy.exit("Exit", "Long", stop=stop_level, limit=target_level)
